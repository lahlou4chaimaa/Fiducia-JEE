' --- MODULE 1 : AUTHENTIFICATION & RÔLES ---
abstract class Utilisateur {
  -id : Long (PK)
  -nom : String
  -prenom : String
  -email : String
  -password : String
}

class Role {
  -id : Long (PK)
  -libelle : String
}

class PersonnelAdministratif {
  -service : String
  -poste : String
}

class CandidatDoctorant {
  -totalCreditsCumules : Double
}

class DirecteurThese {
  -grade : String
  -laboratoire : String
  -specialite : String
  -habilitéDirigerRecherche : boolean
}

' --- MODULE 2 : PROCESSUS D'INSCRIPTION ---
class Campagne {
  -id : Long (PK)
  -libelle : String
  -dateDebut : LocalDate
  -dateFin : LocalDate
  -active : boolean
  +isOuverte() : boolean
}

class DossierInscription {
  -id : Long (PK)
  -sujetThèse : String
  -laboratoire : String
  -cvPath : String
  -diplomePath : String
  -lettreMotivationPath : String
  -collaboration : String
  -statut : StatutDossier
  -dateCreation : LocalDateTime
  -dateDerniereModification : LocalDateTime
  -avisDirecteur : String
  -commentaireDirecteur : String
  -remarquesAdmin : String
}

enum StatutDossier {
  BROUILLON
  SOUMIS
  EN_ATTENTE_DIRECTEUR
  EN_ATTENTE_ADMIN
  VALIDE
  REJETE
}

' --- MODULE 3 : SOUTENANCE ---
class DemandeSoutenance {
  -id : Long (PK)
  -nbArticlesQ1Q2 : int
  -nbConferences : int
  -heuresFormationCompletees : int
  -demandeManuscritePath : String
  -rapportThesePath : String
  -rapportAntiPlagiatPath : String
  -rapportPublicationsCommunicationsPath : String
  -copiesAttestationsFormationsPath : String
  -membresJuryProposes : List<String>
  -dateSoutenance : LocalDateTime
  -lieu : String
  -heure : String
  -statut : StatutSoutenance
  -dateCreation : LocalDateTime
}

enum StatutSoutenance {
  SOUMISE_EN_ATTENTE_JURY
  JURY_PROPOSE
  AUTORISEE_ET_PLANIFIEE
}

' --- HÉRITAGE (InheritanceType.JOINED) ---
Utilisateur <|-- CandidatDoctorant
Utilisateur <|-- DirecteurThese
Utilisateur <|-- PersonnelAdministratif

' --- RELATIONS (Mappage JPA) ---
Utilisateur "0..*" -- "0..*" Role : user_roles

Campagne "0..*" -- "0..*" DirecteurThese : campagne_directeurs
Campagne "1" -- "0..*" DossierInscription : contient

DossierInscription "0..1" -- "1" CandidatDoctorant : lié à
DossierInscription "0..*" -- "0..1" DirecteurThese : est rattaché à

DemandeSoutenance "0..1" -- "1" CandidatDoctorant : doctorant_id
DemandeSoutenance "0..*" -- "0..1" DirecteurThese : directeur_id

' --- NOTES TECHNIQUES ---
note right of Utilisateur
  **Sécurité & Persistance** :
  Stratégie d'héritage : JOINED
  Relation @ManyToMany avec Role
end note

note bottom of DemandeSoutenance
  **Règles de validation métiers** :
  • Heures de formation >= 200
  • Membres Jury stockés en @ElementCollection
