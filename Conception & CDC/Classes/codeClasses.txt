title Fiducia - Diagramme de Classes

' --- MODULE 1 : AUTHENTIFICATION & RÔLES ---
abstract class Utilisateur {
  -id : Long (PK)
  -nom : String
  -prenom : String
  -email : String (Unique)
  -password : String (Argon)
  -role : Role
  +seConnecter()
}

enum Role {
  ADMIN
  CANDIDAT
  DIRECTEUR
  PERSONNEL_ADMIN
}

class CandidatDoctorant {
  -totalCredits : Double
  -derniereInscription : Date
  +creerCompte()
  +deposerManuscrit()
}

class DirecteurThese {
  -specialite : String
  -grade : String
  +donnerAvis(dossierId: Long)
  +proposerJury()
}

' --- MODULE 2 : PROCESSUS D'INSCRIPTION ---
class Campagne {
  -id : Long
  -libelle : String
  -dateOuverture : Date
  -dateFermeture : Date
  -estOuverte : boolean
}

class DossierInscription {
  -id : Long
  -sujetThese : String
  -avisDirecteur : Statut (PENDING/VALIDATED)
  -validationAdmin : Statut
  -dateSoumission : DateTime
  +notifierActeurs()
}

class PieceJustificative {
  -id : Long
  -libelle : String
  -cheminFichier : String (Storage Path)
  -format : String (PDF/JPG)
  +verifierValidite()
}

' --- MODULE 3 : SOUTENANCE ---
class DemandeSoutenance {
  -id : Long
  -nbPublications : int
  -creditsFormation : Double
  -statut : StatutSoutenance
  +verifierChecklist()
}

class Jury {
  -id : Long
  -nomComplet : String
  -qualite : String (Rapporteur/Examinateur)
  -etablissement : String
}

' --- HERITAGE ---
Utilisateur <|-- CandidatDoctorant
Utilisateur <|-- DirecteurThese

' --- RELATIONS ---
Campagne "1" o-- "0..*" DossierInscription : cadre
CandidatDoctorant "1" ||-- "0..*" DossierInscription : soumet
DossierInscription "1" *-- "1..*" PieceJustificative : contient
DossierInscription "0..*" -- "1" DirecteurThese : est rattaché à

CandidatDoctorant "1" ||-- "0..1" DemandeSoutenance : initie
DemandeSoutenance "1" -- "3..6" Jury : composé de
